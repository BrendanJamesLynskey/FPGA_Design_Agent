#!/bin/bash
# run_one.sh <A|B> <task> <tag>   -- one brief-167 agent run (see prereg.md). Logs to _scratch_167/logs/<tag>.log
set -u
S=/home/brendan/Claude_sandbox/_scratch_167; H=/home/brendan/Claude_sandbox/_handoff
arm=$1; task=$2; tag=$3
case $arm in A) W=/home/brendan/Claude_sandbox/_worktrees/167-armA;; B) W=/home/brendan/Claude_sandbox/_worktrees/167-vhdl-rerun;; esac
mkdir -p $S/logs; L=$S/logs/$tag.log
[ -e $S/STOP ] && { echo "STOP flag set" | tee -a $L; exit 3; }
ver=$($S/bin/claude --version | cut -d' ' -f1)
[ "$ver" = 2.1.321 ] || { echo "PREFLIGHT_FAIL $tag: CLI $ver" | tee -a $L; touch $S/STOP; exit 4; }
pct=$($H/usage_check.sh 2>&1 | grep 'Current week (all models)' | grep -o '[0-9]*% used' | head -1 | tr -d '%used ')
echo "$(date '+%F %T') $tag arm=$arm task=$task weekly=${pct}%" | tee -a $L
[ "${pct:-100}" -ge 97 ] && { echo "usage >= 97" | tee -a $L; touch $S/STOP; exit 5; }
text="167: agent run $tag (VHDL e2e, Vivado elaboration only, no impl), $(date +%H:%M), expected end $(date -d '+60 min' +%H:%M), peak 6G"
until lock=$($H/take_heavy_slot.sh "$text" 2>/dev/null); do sleep 0.2; [ -e $S/STOP ] && exit 3; done
echo "$(date '+%F %T') slot $lock taken" | tee -a $L
cd $W
export PATH=$S/bin:$PATH FDA_KNOWLEDGE=/home/brendan/Claude_sandbox/_worktrees/167-kb FDA_PROMPTS_DIR=/home/brendan/Claude_sandbox/_worktrees/167-kb/prompts
export FDA_VIVADO_MAX=2
timeout -k 30 5400 systemd-run --user --scope -q -p MemoryMax=8G nice -n 15 \
  /home/brendan/Claude_sandbox/FPGA_Design_Agent/.venv/bin/python -m agent run benchmark/tasks/$task/spec.md --hdl vhdl --stop-after 2 \
  --verif-level V1 --harness claude_code --cc-model claude-haiku-5-5 --cc-effort medium --max-usd 0.75 --name ${tag} >> $L 2>&1
echo "$(date '+%F %T') exit $?" | tee -a $L
case "$(cat $lock)" in "167:"*) rm -f $lock;; esac
